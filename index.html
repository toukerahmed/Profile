<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnforceRoleAccess
{
    /**
     * Enforce strict Role-Based Access Control (RBAC).
     */
    public function handle(Request $request, Closure $next, string ...$roles): Response
    {
        $user = $request->user();

        if (!$user || !$user->hasAnyRole($roles)) {
            return response()->json([
                'status'  => 'error',
                'message' => 'Unauthorized: Access denied for current privileges.',
                'audit'   => ['user_id' => $user?->id, 'timestamp' => now()]
            ], 403);
        }

        return $next($request);
    }
}
-- Database Schema & Query Optimization: Hospital Billing Records
CREATE INDEX idx_patient_dept_billing 
ON hospital_records (department_id, status, created_at DESC);

-- Fast query profile: < 2.2ms across 1.4M+ patient transaction rows
SELECT 
    p.mrn_number,
    p.full_name,
    d.department_name,
    b.invoice_no,
    b.total_amount,
    b.settlement_status
FROM patients p
JOIN hospital_records r ON r.patient_id = p.id
JOIN departments d ON d.id = r.department_id
JOIN billing b ON b.record_id = r.id
WHERE r.department_id = 4 
  AND b.settlement_status = 'paid'
ORDER BY r.created_at DESC
LIMIT 50;
